#include <GL/glut.h>
#include <iostream>
using namespace std;

// Global variables used by all algorithms
int choice;
int x1_, y1_, x2_, y2_;
int xc, yc, r;

// ------------------- DDA Line Algorithm -------------------
void drawDDA(int x1, int y1, int x2, int y2) {
    float dx = x2 - x1;
    float dy = y2 - y1;

    int steps = (abs(dx) > abs(dy)) ? abs(dx) : abs(dy);

    float Xinc = dx / steps;
    float Yinc = dy / steps;

    float X = x1;
    float Y = y1;

    glBegin(GL_POINTS);
    for (int i = 0; i <= steps; i++) {
        glVertex2i(X, Y);
        X += Xinc;
        Y += Yinc;
    }
    glEnd();
}

// ------------------- Midpoint Line Algorithm -------------------
void drawMidpointLine(int x1, int y1, int x2, int y2) {
    int dx = abs(x2 - x1);
    int dy = abs(y2 - y1);
    int p = 2 * dy - dx;

    int x, y;

    if (x1 > x2) {
        x = x2;
        y = y2;
        x2 = x1;
        y2 = y1;
    } else {
        x = x1;
        y = y1;
    }

    glBegin(GL_POINTS);
    glVertex2i(x, y);

    while (x < x2) {
        x++;
        if (p < 0) {
            p += 2 * dy;
        } else {
            y++;
            p += 2 * (dy - dx);
        }
        glVertex2i(x, y);
    }
    glEnd();
}

// ------------------- Midpoint Circle Algorithm -------------------
void drawMidpointCircle(int xc, int yc, int r) {
    int x = 0, y = r;
    int p = 1 - r;

    glBegin(GL_POINTS);

    while (x <= y) {
        // 8 symmetric points
        glVertex2i(xc + x, yc + y);
        glVertex2i(xc - x, yc + y);
        glVertex2i(xc + x, yc - y);
        glVertex2i(xc - x, yc - y);
        glVertex2i(xc + y, yc + x);
        glVertex2i(xc - y, yc + x);
        glVertex2i(xc + y, yc - x);
        glVertex2i(xc - y, yc - x);

        if (p < 0) {
            p += 2 * x + 3;
        } else {
            p += 2 * (x - y) + 5;
            y--;
        }
        x++;
    }

    glEnd();
}

// ------------------- Display Function -------------------
void display() {
    glClear(GL_COLOR_BUFFER_BIT);

    glColor3f(0, 0, 0);  // Black points

    if (choice == 1)
        drawDDA(x1_, y1_, x2_, y2_);
    else if (choice == 2)
        drawMidpointLine(x1_, y1_, x2_, y2_);
    else if (choice == 3)
        drawMidpointCircle(xc, yc, r);

    glFlush();
}

// ------------------- Setup -------------------
void init() {
    glClearColor(1, 1, 1, 1); // White background
    glColor3f(0, 0, 0);       // Black drawing
    gluOrtho2D(0, 500, 0, 500);
}

// ------------------- Main Function -------------------
int main(int argc, char** argv) {

    cout << "Choose algorithm:\n";
    cout << "1. DDA Line\n";
    cout << "2. Midpoint Line\n";
    cout << "3. Midpoint Circle\n";
    cout << "Enter choice: ";
    cin >> choice;

    if (choice == 1 || choice == 2) {
        cout << "Enter x1 y1: ";
        cin >> x1_ >> y1_;
        cout << "Enter x2 y2: ";
        cin >> x2_ >> y2_;
    }
    else if (choice == 3) {
        cout << "Enter center (xc yc): ";
        cin >> xc >> yc;
        cout << "Enter radius: ";
        cin >> r;
    }
    else {
        cout << "Invalid choice!";
        return 0;
    }

    glutInit(&argc, argv);
    glutInitWindowSize(600, 600);
    glutInitWindowPosition(100, 100);
    glutCreateWindow("DDA / Midpoint Line / Midpoint Circle");
    init();
    glutDisplayFunc(display);
    glutMainLoop();

    return 0;
}
